<?php
session_start();
include "../db_conn.php";

if (!isset($_SESSION['user_id'])) {
    header("Location: ../login.php");
    exit;
}

$user_id = $_SESSION['user_id'];

// CSRF
if (empty($_POST['csrf_token']) || !hash_equals($_SESSION['csrf_token'] ?? '', $_POST['csrf_token'])) {
    header("Location: ../change-password.php?error=" . urlencode("Invalid request."));
    exit;
}

$current  = $_POST['current_password'] ?? '';
$new      = $_POST['new_password'] ?? '';
$confirm  = $_POST['confirm_password'] ?? '';

if (empty($current) || empty($new) || empty($confirm)) {
    header("Location: ../change-password.php?error=" . urlencode("All fields are required."));
    exit;
}

if (strlen($new) < 8 || !preg_match('/[A-Z]/', $new) || !preg_match('/[0-9]/', $new)) {
    header("Location: ../change-password.php?error=" . urlencode("Password must be 8+ chars with uppercase and number."));
    exit;
}

if ($new !== $confirm) {
    header("Location: ../change-password.php?error=" . urlencode("New passwords do not match."));
    exit;
}

// Get current password hash
$stmt = $conn->prepare("SELECT password FROM admin WHERE id = ? LIMIT 1");
$stmt->execute([$user_id]);
$user = $stmt->fetch();

if (!$user || !password_verify($current, $user['password'])) {
    header("Location: ../change-password.php?error=" . urlencode("Current password is incorrect."));
    exit;
}

// Update with new bcrypt hash
$hashed = password_hash($new, PASSWORD_BCRYPT, ['cost' => 12]);
$update = $conn->prepare("UPDATE admin SET password = ? WHERE id = ?");
$update->execute([$hashed, $user_id]);

unset($_SESSION['csrf_token']);

header("Location: ../change-password.php?success=" . urlencode("Password changed successfully!"));
exit;
